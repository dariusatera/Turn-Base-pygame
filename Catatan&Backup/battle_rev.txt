def Slayer(screen):
    slayer_pos = (WIDTH // 8, HEIGHT // 2 + 200)  
    slayer_radius = 40
    SLAYER_COLOR = (WHITE)
    
    bar_pos = (WIDTH // 8 + 20, HEIGHT // 2 + 200 + slayer_radius + 10)  # Posisi bar HP
    bar_hihght = 20
    bar_width = 40
    

    running = True
    while running:
        for event in pygame.event.get():
            if event.type == pygame.QUIT:
                running = False # Keluar dari loop dan program

        screen.fill(BLACK)
        pygame.draw.rect(screen, WHITE, (bar_pos[0], bar_pos[1], bar_width, bar_hihght))  # Menggambar bar HP
        pygame.draw.circle(screen, SLAYER_COLOR, slayer_pos, slayer_radius)
        pygame.display.update()  # <-- Pindahkan ke dalam loop

    return "QUIT"
